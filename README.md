```mermaid
flowchart TD
    A["USER INPUT"] --> B["APPLICATION API"]
    B --> C["CONTEXT RETRIEVAL"]

    D --> E{"TOOL REQUIRED?"}
    E -- YES --> F["TOOL EXECUTION"]
    F --> G["RESULT VALIDATION"]
    G --> D
    E -- NO --> H["RESPONSE GENERATION"]
    H --> I["CLIENT RESPONSE"]
```
